--
DOCKER COMMANDS
--
docker run --name product-service-postgres \
  -e POSTGRES_PASSWORD=productservice \
  -p 5432:5432 \
  -v postgres_data:/var/lib/postgresql/data \
  -d postgres

-------------------------------------------------------
CREATES DOCKER VOLUME
-------------------------------------------------------
docker volume create product_data
docker volume create payment_data

-------------------------------------------------------
DOCKER RUN
-------------------------------------------------------
docker run --name product-service-postgres -e POSTGRES_PASSWORD=productservice -p 5432:5432 -v product_data:/var/lib/postgresql/18/main -d postgres
docker run --name order-service-postgres -e POSTGRES_PASSWORD=orderservice -p 5433:5432 -v order data:/var/lib/postgresql/18/main -d postgres

docker run --name product-service-postgres -e POSTGRES_PASSWORD=productservice -p 5432:5432 -v product_data:/var/lib/postgresql/18/main -d postgres:alpine
docker run --name order-service-postgres -e POSTGRES_PASSWORD=orderservice -p 5433:5432 -v order_data:/var/lib/postgresql/18/main -d postgres:alpine
docker run --name payment-service-postgres -e POSTGRES_PASSWORD=paymentservice -p 5433:5432 -v payment_data:/var/lib/postgresql/18/main -d postgres:alpine


-------------------------------------------------------
CLI
-------------------------------------------------------
docker exec -it product-service-postgres psql -U postgres
docker exec -it order-service-postgres psql -U postgres
